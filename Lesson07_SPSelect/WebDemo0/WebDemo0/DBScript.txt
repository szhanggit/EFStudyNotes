(
     ID int primary key identity,
     Name nvarchar(50),
     Gender nvarchar(50),

Create procedure InsertSPEmployee
@Name nvarchar(50),
@Gender nvarchar(50),
@Salary int
as
Begin
     Insert into SPEmployees values (@Name, @Gender, @Salary)   
End
Go

Create procedure UpdateSPEmployee
@ID int,
@Name nvarchar(50),
@Gender nvarchar(50),
@Salary int
as
Begin
     Update SPEmployees Set [Name] = @Name, Gender = @Gender,
     Salary = @Salary
     where ID = @ID
End
Go

Create procedure DeleteSPEmployee
@ID int
as
Begin
     Delete from SPEmployees where ID = @ID
End
Go
