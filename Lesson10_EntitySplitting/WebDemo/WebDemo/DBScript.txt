Create table SplitEmployees
(
     EmployeeID int primary key identity,
     FirstName nvarchar(50),
     LastName nvarchar(50),
     Gender nvarchar(50)
)
GO

Create table EmployeeContactDetails
(
     EmployeeID int primary key,
     Email nvarchar(50),
     Mobile nvarchar(50),
     LandLine nvarchar(50),
	 Constraint fk_pk_empid FOREIGN KEY(EmployeeId) references SplitEmployees(EmployeeID)
)
GO
