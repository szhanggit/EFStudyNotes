Create Table BridgeCourses
(
     CourseID int identity primary key,
     CourseName nvarchar(50)
)
GO

Create Table BridgeStudents
(
     StudentID int identity primary key,
     StudentName nvarchar(50)
)
GO

Create Table BridgeStudentCourses
(
     StudentID int not null foreign key references BridgeStudents(StudentID),
     CourseID int not null foreign key references BridgeCourses(CourseID),
     EnrolledDate DateTime,
     primary key (StudentID, CourseID)
)
GO
