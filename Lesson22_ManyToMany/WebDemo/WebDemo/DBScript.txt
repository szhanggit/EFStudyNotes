Create Table MTMCourses
(
     CourseID int identity primary key,
     CourseName nvarchar(50)
)
GO

Create Table MTMStudents
(
     StudentID int identity primary key,
     StudentName nvarchar(50)
)
GO

Create Table MTMStudentCourses
(
     StudentID int not null foreign key references MTMStudents(StudentID),
     CourseID int not null foreign key references MTMCourses(CourseID)
     primary key (StudentID, CourseID)
)
GO
